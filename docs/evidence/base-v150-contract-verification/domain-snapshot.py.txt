import hashlib,json,pathlib,sys
root=pathlib.Path.cwd();target=root/'var/cross-language-v150'/sys.argv[1]
files=['src/domain/publication.ts','src/domain/shards.ts','src/domain/record-contract.ts','src/domain/age.ts',
       'src/domain/schema-pin.ts','src/domain/errors.ts','src/domain/normalize.ts','src/domain/visibility.ts',
       'src/domain/bundle-file.ts','src/db/client.ts','src/db/schema.ts','src/shared/contracts.ts']
payload={p:(root/p).read_bytes() for p in files};hashes={p:hashlib.sha256(v).hexdigest() for p,v in payload.items()}
for p,v in payload.items():
 path=target/p;path.parent.mkdir(parents=True,exist_ok=True);path.write_bytes(v)
assert all((root/p).read_bytes()==v for p,v in payload.items()),'Domain code changed while capturing snapshot'
digest=hashlib.sha256(json.dumps(hashes,sort_keys=True,separators=(',',':')).encode()).hexdigest()
(target/'snapshot.json').write_text(json.dumps({'files':hashes,'sha256':digest},indent=2)+'\n');print(target)
